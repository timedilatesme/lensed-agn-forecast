# <band>: every band of the catalog (LSST u g r i z y, Roman F062 F087 F106 F129 F158 F184 F146 F213, Euclid VIS Y J H)
# <image#>: image slot 1-5, in SLSim's order (by arrival time); NaN when the system has fewer images
# Position angles: '*_position_angle_deg' is 0.5 * arctan2(e2, e1) in SLSim's frame; lenstronomy (and so the image positions)
# uses (-e1, e2), given by the '*_position_angle_lenstronomy_deg' columns.

# --- System Properties ---
# 'lens_id',                                     # (row index in the catalog + SLSim's Lens.generate_id, unique within a catalog)
# 'z_S', 'z_D',
# 'theta_E_arcsec',                              # (Einstein radius, Lens.einstein_radius, includes the external-convergence rescale)
# 'theta_E_lens_model_arcsec',                   # (Einstein radius of the EPL in the lenstronomy lens model, no rescale)
# 'max_time_delay_days',
# 'num_images',
# 'max_image_separation_arcsec',                 # (largest separation between two images)
# 'om10_selected',                               # (Oguri & Marshall 2010 [OM10] cuts: 2nd-brightest image of doubles/triples, 3rd-brightest
#                                                #  of quads, i <= 23.3; separation >= 0.5"; doubles need a flux ratio >= 0.1)

# --- Point Source (AGN) ---
# 'x_ps_position_arcsec',
# 'y_ps_position_arcsec',
# 'unlensed_ps_mag_<band>'                       # (magnitude of the unlensed quasar in each band; qsogen SED, no host)
# 'faintest_image_ps_mag_<band>'                 # (magnitude of the faintest lensed quasar image in each band)
# 'third_brightest_image_ps_mag_<band>'          # (magnitude of the third brightest lensed quasar image; NaN with fewer than 3 images)
# 'second_brightest_image_ps_mag_<band>'         # (magnitude of the second brightest lensed quasar image in each band)
# 'brightest_image_ps_mag_<band>'                # (magnitude of the brightest lensed quasar image in each band)
# 'lensed_ps_mag_<band>_<image#>'                # (magnitudes of the lensed quasar images in each band)
# 'time_delay_days_<image#>',                    # (arrival time of each image with respect to the straight line arrival time)
# 'ps_magnification_<image#>',                   # (signed magnification of each lensed quasar image)
# 'x_ps_image_positions_arcsec_<image#>',
# 'y_ps_image_positions_arcsec_<image#>',
# 'M_i',                                         # (absolute i magnitude M_i(z=2) from the OM10 quasar luminosity function with
#                                                #  the Richards et al. 2005 break and Richards et al. 2006 K-correction; SLSim)
# 'log_bolometric_luminosity',                   # (log10 L_bol [erg/s] from M_i via L3000 (zero point fitted to Wu & Shen 2022)
#                                                #  and L_bol = 5.15 L3000 (Richards et al. 2006); SLSim)
# 'black_hole_mass_exponent',                    # (log10 M_BH [M_sun], SLSim host matching conditioned on M_i and the host: red hosts
#                                                #  M-sigma (Kormendy & Ho 2013), blue hosts M-M* (Reines & Volonteri 2015))
# 'agn_disk_eddington_ratio',                    # (Eddington ratio, linear; lognormal prior, mean log -1.15, width 0.30 dex,
#                                                #  fitted to Wu & Shen 2022; SLSim host matching)
# 'black_hole_spin',                             # (dimensionless spin parameter of the black hole; 0)
# 'agn_disk_inclination_angle',                  # (disk inclination w.r.t. the observer in degrees, isotropic up to 60 deg;
#                                                #  type-1 AGN assumption motivated by the unified model, Urry & Padovani 1995)
# 'agn_disk_position_angle_deg',                 # (disk position angle on the sky in degrees, uniform)
# 'tau_rest_days',                               # (DRW damping timescale, rest frame, drawn about Suberlak et al. 2021 with
#                                                #  0.15 dex scatter, correlation 0.16 with SF_inf; the scatter is approximate,
#                                                #  from MacLeod et al. 2010, not a published value)
# 'sf_inf_mag',                                  # (DRW asymptotic structure function SF_inf at rest 4000 A, Suberlak et al. 2021,
#                                                #  0.04 dex scatter, drawn jointly with tau)
# 'sf_inf_mag_<band>',                           # (SF_inf at the band's rest-frame wavelength, Suberlak et al. 2021 wavelength scaling)
# 'log10_tau_suberlak21', 'log10_sf_inf_suberlak21', # (Suberlak et al. 2021 means, no scatter; tau in days, SF_inf in mag)
# 'log10_tau_macleod10', 'log10_sf_inf_macleod10',   # (MacLeod et al. 2010 means)
# 'log10_tau_burke21',                           # (Burke et al. 2021 mean, tau = 107 d (M_BH / 1e8 M_sun)^0.38)
# 'break_timescale_days',                        # (observer-frame DRW break, 2 pi tau_rest (1 + z_S))

# --- Host Galaxy (Source) Light [SINGLE SERSIC] ---
# 'host_galaxy_type',                            # (red or blue; red hosts get n_sersic = 4 and M_BH from M-sigma, blue n = 1 and M-M*)
# 'host_stellar_mass',                           # (M_sun, SkyPy)
# 'host_velocity_dispersion',                    # (km/s, abundance matched to the SDSS velocity dispersion function of
#                                                #  Bernardi et al. 2010 at z < 0.5)
# 'x_host_position_arcsec',
# 'y_host_position_arcsec',
# 'unlensed_host_mag_<band>' for all bands,
# 'lensed_host_mag_<band>' for all bands,
# 'host_light_R_eff_arcsec',
# 'host_light_n_sersic',
# 'host_light_e1',
# 'host_light_e2',
# 'host_light_ellipticity',            # (computes as sqrt(e1^2 + e2^2))
# 'host_light_axis_ratio',             # (q_host = (1 - e_host)/(1 + e_host))
# 'host_light_position_angle_deg',     # (computes as 0.5 * arctan2(e2, e1))
# 'host_light_position_angle_lenstronomy_deg',

# --- Deflector Light (Lens Galaxy) [SINGLE SERSIC] ---
# 'deflector_mag_<band>' for all bands,
# 'deflector_light_R_eff_arcsec',
# 'deflector_light_n_sersic',           # (4 for red, 1 for blue deflectors)
# 'deflector_light_e1',
# 'deflector_light_e2',
# 'deflector_light_ellipticity',        # (computes as sqrt(e1^2 + e2^2))
# 'deflector_light_position_angle_deg', # (allows light/mass misalignment, computed as 0.5 * arctan2(e2, e1))
# 'deflector_light_position_angle_lenstronomy_deg',
# 'deflector_light_axis_ratio',         # (q_light = (1 - e_light)/(1 + e_light))
# 'x_deflector_light_position_arcsec',  # (if light center differs from mass center)
# 'y_deflector_light_position_arcsec',  # (if light center differs from mass center)
# 'deflector_surface_brightness_<band>_<image#>', # (deflector surface brightness at each image, mag/arcsec^2)

# --- Deflector Mass (Lens Galaxy) [PEMD + SHEAR] ---
# 'x_deflector_mass_position_arcsec',
# 'y_deflector_mass_position_arcsec',
# 'deflector_stellar_mass',             # (Chabrier M* implied by sigma through the Auger et al. 2010 sigma-M* relation
#                                       #  (Table 2, fit without scatter): log10 sigma = 0.24 log10(M*/1e11) + 2.32; the mass
#                                       #  behind kappa_star, so the stars are consistent with the lens mass)
# 'deflector_stellar_mass_slsim',       # (SkyPy stellar mass that SLSim uses; behind kappa_star_slsim)
# 'deflector_pl_slope',                 # (gamma_pl, drawn from N(2.078, 0.16), the SLACS slope distribution of Auger et al. 2010)
# 'deflector_velocity_dispersion',      # (sigma_v_D, abundance matched to the SDSS velocity dispersion function of
#                                       #  Bernardi et al. 2010 at z < 0.5; sets the EPL mass)
# 'deflector_mass_e1',                  # (e1_mass)
# 'deflector_mass_e2',                  # (e2_mass)
# 'deflector_mass_ellipticity',         # (e_mass = sqrt(e1_mass^2 + e2_mass^2))
# 'deflector_mass_axis_ratio',          # (q_mass = (1 - e_mass)/(1 + e_mass))
# 'deflector_mass_position_angle_deg',  # (phi_mass = 0.5 * arctan2(e2_mass, e1_mass))
# 'deflector_mass_position_angle_lenstronomy_deg',
# 'external_shear_gamma1',              # (gamma1 = gamma_ext * cos(2*phi_ext); lenstronomy's gamma1 is -gamma1)
# 'external_shear_gamma2',              # (gamma2 = gamma_ext * sin(2*phi_ext))
# 'external_shear_amplitude',           # (gamma_ext = sqrt(gamma1^2 + gamma2^2))
# 'external_shear_position_angle_deg',  # (phi_ext = 0.5 * arctan2(gamma2, gamma1))
# 'external_shear_position_angle_lenstronomy_deg',
# 'external_convergence',               # (kappa_ext)

# --- Microlensing Parameters (Per Image) ---
# 'kappa_star_<image#>'                 # (stellar convergence at the image: deflector light profile x deflector_stellar_mass;
#                                       #  capped just below kappa_tot)
# 'kappa_star_capped_<image#>'          # (1 where kappa_star was capped, else 0)
# 'kappa_star_slsim_<image#>'           # (SLSim's stellar convergence, with deflector_stellar_mass_slsim; capped as above)
# 'kappa_star_slsim_capped_<image#>'    # (1 where kappa_star_slsim was capped, else 0)
# 'kappa_tot_<image#>'                  # (total convergence at the image, includes kappa_ext)
# 'shear_<image#>'                      # (total shear amplitude at the image, includes the external shear)
# 'shear_angle_<image#>'                # (total shear angle in radians, arctan2(gamma2, gamma1) = 2 phi, lenstronomy frame)
